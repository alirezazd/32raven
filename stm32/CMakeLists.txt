# Every first-party Python script runs through `uv run --script`. Each script
# declares its own dependencies inline (PEP 723), so the build resolves them
# itself instead of trusting whatever happens to be installed in the ambient
# interpreter — which differs between the Docker image, a host install, and the
# ESP-IDF virtualenv. uv caches after the first run.
find_program(UV_EXECUTABLE NAMES uv REQUIRED)
set(UV_RUN ${UV_EXECUTABLE} run --quiet --script)

find_program(CLANG_FORMAT_EXECUTABLE NAMES clang-format)

# Validate the freshly-generated stm32_config.hpp against ST's silicon pin /
# AF data as part of header generation. Catches typo'd alternate functions,
# pins not on the F407V package, same pin claimed twice, and unreferenced /
# unknown board::k* names. Failure aborts the build — bad pin maps brick
# boards silently otherwise.
set(STM32_PIN_CHECKER
    "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/lint/check_pinmap.py")
set(ERROR_CODE_HPP
    "${CMAKE_CURRENT_SOURCE_DIR}/../libs/inc/error_code.hpp")

# Conditional clang-format step — empty when the binary isn't on PATH so the
# main custom_command stays single-form regardless. CMake skips empty COMMAND
# entries cleanly.
set(_FMT_CONFIG_CMD)
set(_FMT_EE_CMD)
if(CLANG_FORMAT_EXECUTABLE)
    set(_FMT_CONFIG_CMD COMMAND ${CLANG_FORMAT_EXECUTABLE} -i
        ${STM32_RUNTIME_CONFIG_HPP} ${STM32_LIMITS_HPP})
    set(_FMT_EE_CMD COMMAND ${CLANG_FORMAT_EXECUTABLE} -i ${STM32_EE_SCHEMA_HPP})
endif()

add_custom_command(
    OUTPUT ${STM32_RUNTIME_CONFIG_HPP} ${STM32_LIMITS_HPP}
    COMMAND ${UV_RUN} ${STM32_CONFIG_GENERATOR}
    --kconfig ${STM32_KCONFIG}
    --config ${STM32_MASTER_CONFIG}
    --runtime-out ${STM32_RUNTIME_CONFIG_HPP}
    --limits-out ${STM32_LIMITS_HPP}
    ${_FMT_CONFIG_CMD}
    COMMAND ${UV_RUN} ${STM32_PIN_CHECKER}
    DEPENDS ${STM32_KCONFIG} ${STM32_MASTER_CONFIG}
    ${STM32_CONFIG_GENERATOR}
    ${STM32_RUNTIME_CONFIG_TEMPLATE}
    ${STM32_LIMITS_TEMPLATE}
    ${STM32_PIN_CHECKER}
    ${ERROR_CODE_HPP}
    COMMENT "Generating + validating STM32 configuration headers"
    VERBATIM)

add_custom_target(generate_stm32_config
    DEPENDS ${STM32_RUNTIME_CONFIG_HPP}
    ${STM32_LIMITS_HPP})
add_dependencies(${CMAKE_PROJECT_NAME} generate_stm32_config)

add_custom_command(
    OUTPUT ${STM32_EE_SCHEMA_HPP}
    COMMAND ${UV_RUN} ${STM32_EE_SCHEMA_GENERATOR}
    --config ${STM32_EE_CONFIG}
    --out ${STM32_EE_SCHEMA_HPP}
    ${_FMT_EE_CMD}
    DEPENDS ${STM32_EE_CONFIG} ${STM32_EE_SCHEMA_GENERATOR}
    ${STM32_EE_SCHEMA_TEMPLATE}
    COMMENT "Generating EEPROM schema header"
    VERBATIM)

add_custom_target(generate_ee_schema DEPENDS ${STM32_EE_SCHEMA_HPP})
add_dependencies(${CMAKE_PROJECT_NAME} generate_ee_schema)

# Eigen runtime knobs — this firmware has no heap (no operator new, no
# malloc by policy). Anything that triggers a dynamic allocation inside
# Eigen is a bug, not a fallback.
#   EIGEN_NO_MALLOC       — every Eigen heap-allocation path is gated
#                           behind an eigen_assert that always fires.
#                           Acts as a compile-time guarantee that the
#                           only Eigen types we instantiate are fixed-
#                           size (today: Vector3f, Quaternionf).
#   EIGEN_NO_AUTOMATIC_RESIZING — refuse silent resize on assignment.
#                           Reinforces "every matrix size is a static
#                           property" with a hard error instead of a
#                           silent realloc.
#   EIGEN_RUNTIME_NO_MALLOC — also enables `set_is_malloc_allowed(bool)`
#   EIGEN_STACK_ALLOCATION_LIMIT=0 — disable the per-temporary stack
#                           limit (we don't need the heuristic; fixed-
#                           size types live entirely on the stack /
#                           inside the owning object anyway).
target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE
    EIGEN_NO_MALLOC
    EIGEN_NO_AUTOMATIC_RESIZING
    EIGEN_RUNTIME_NO_MALLOC
    EIGEN_STACK_ALLOCATION_LIMIT=0
)

target_compile_options(${CMAKE_PROJECT_NAME} PRIVATE
    $<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>
    -fno-exceptions
    -ffunction-sections
    -fdata-sections
    -Wdouble-promotion
    -Wunused
    -Werror=switch
    -Wno-psabi # Suppress Eigen's Matrix warnings
)

target_link_options(${CMAKE_PROJECT_NAME} PRIVATE
    -Wl,--gc-sections
    # Route vsnprintf / snprintf through nanoprintf
    # (stm32/Core/printf_impl.cpp). newlib's nano-svfprintf
    # pulled in _malloc_r / _free_r / _sbrk for unbounded conversions;
    # nanoprintf is allocation-free, so those symbols leave the binary
    # entirely once the linker stops needing newlib's printf object.
    -Wl,--wrap=vsnprintf
    -Wl,--wrap=snprintf
)

# Keep printf_impl.cpp out of LTO. The linker's --wrap (above) rebinds calls to
# __wrap_vsnprintf/__wrap_snprintf at final link, but LTO doesn't model --wrap,
# so it internalizes those wrappers as "uncalled" and the references dangle.
# Compiling this one tiny TU without LTO leaves the __wrap_* symbols as real,
# linker-visible symbols. (No LTO loss — it's just the nanoprintf shim.)
set_source_files_properties(Core/printf_impl.cpp PROPERTIES
    COMPILE_OPTIONS "-fno-lto")
