# The generator runs through `uv run --script`, which resolves the script's own
# PEP 723 dependencies into an isolated env. This replaces the previous dance of
# hunting for a system python3 outside the ESP-IDF virtualenv: IDF sets
# Python3_EXECUTABLE to its own venv, which has no kconfiglib or jinja2, so the
# build had to escape it to reach the apt-installed packages. uv makes the
# generator independent of whichever interpreter IDF or CMake picked.
find_program(UV_EXECUTABLE NAMES uv REQUIRED)
set(UV_RUN ${UV_EXECUTABLE} run --quiet --script)

find_program(CLANG_FORMAT_EXECUTABLE NAMES clang-format)

# Conditional clang-format step — empty when the binary isn't on PATH so the
# main custom_command stays single-form regardless. CMake skips empty COMMAND
# entries cleanly.
set(_FMT_CMD)
if(CLANG_FORMAT_EXECUTABLE)
    set(_FMT_CMD COMMAND ${CLANG_FORMAT_EXECUTABLE} -i
                 ${ESP32_RUNTIME_CONFIG_HPP} ${ESP32_LIMITS_HPP})
endif()

add_custom_command(
    OUTPUT ${ESP32_RUNTIME_CONFIG_HPP} ${ESP32_LIMITS_HPP}
    COMMAND ${UV_RUN} ${ESP32_CONFIG_GENERATOR}
            --kconfig ${ESP32_KCONFIG}
            --config ${ESP32_MASTER_CONFIG}
            --runtime-out ${ESP32_RUNTIME_CONFIG_HPP}
            --limits-out ${ESP32_LIMITS_HPP}
    ${_FMT_CMD}
    DEPENDS ${ESP32_KCONFIG} ${ESP32_MASTER_CONFIG}
            ${ESP32_CONFIG_GENERATOR}
            ${ESP32_FIRMWARE_VERSION_TOOL}
            ${ESP32_FIRMWARE_VERSION_FILE}
            ${ESP32_CONFIG_TEMPLATE}
            ${ESP32_LIMITS_TEMPLATE}
    COMMENT "Generating ESP32 configuration headers"
    VERBATIM)

add_custom_target(generate_esp32_config
    DEPENDS ${ESP32_RUNTIME_CONFIG_HPP}
    ${ESP32_LIMITS_HPP})
add_dependencies(${COMPONENT_LIB} generate_esp32_config)

# `ARDUINO=100` keeps the Adafruit GFX code happy, and the packed-member warning
# is suppressed because the shared wire/protocol structs intentionally use
# packed layouts.
target_compile_definitions(${COMPONENT_LIB} PRIVATE ARDUINO=100)
target_compile_options(${COMPONENT_LIB} PRIVATE "-Wno-address-of-packed-member")
